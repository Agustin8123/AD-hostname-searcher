cmake_minimum_required(VERSION 3.20)
project(ADHostnameSearcher VERSION 2.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_executable(ADHostnameSearcher WIN32 src/main.cpp src/windows_services.cpp src/app.rc)
target_include_directories(ADHostnameSearcher PRIVATE src)
target_link_libraries(ADHostnameSearcher PRIVATE comctl32 uxtheme gdi32 user32 adsiid activeds ole32 oleaut32 ws2_32 iphlpapi)
if(MINGW)
  target_link_options(ADHostnameSearcher PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
  target_compile_options(ADHostnameSearcher PRIVATE /W4 /utf-8)
  set_property(TARGET ADHostnameSearcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
add_test(NAME core_tests COMMAND core_tests)
add_executable(network_tests tests/network_tests.cpp src/windows_services.cpp)
target_link_libraries(network_tests PRIVATE adsiid activeds ole32 oleaut32 ws2_32 iphlpapi)
if(MINGW)
  target_link_options(core_tests PRIVATE -static)
  target_link_options(network_tests PRIVATE -static)
endif()
add_test(NAME network_loopback COMMAND network_tests)
